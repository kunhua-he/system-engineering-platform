# -*- coding: utf-8 -*-
"""零调用条目三口径现场重算（#69）。只读扫描，不写仓库。"""
import ast, os, re, json
from collections import Counter

根 = "/Users/hekunhua/Documents/Agent/PHP/系统工程平台"
排除目录 = {".git", "工程缓存", "__pycache__", ".venv", "node_modules"}
七大根 = {"支持库", "模块库", "运行核心", "平台控制面", "开发工具", "公共契约", "测试中心"}
六大根 = 七大根 - {"测试中心"}
词模式 = re.compile(r"[A-Za-z_\u4e00-\u9fff][\w\u4e00-\u9fff]*")


def 遍历(根集, 只py):
    表 = []
    for 当前, 子目录表, 文件表 in os.walk(根):
        子目录表[:] = [d for d in 子目录表 if d not in 排除目录]
        部分 = os.path.relpath(当前, 根).split(os.sep)
        if 部分[0] != "." and 部分[0] not in 根集:
            子目录表[:] = []
            continue
        for f in 文件表:
            if 只py and not f.endswith(".py"):
                continue
            表.append(os.path.join(当前, f))
    return sorted(表)


def 文本计数(根集):
    计数 = Counter()
    for p in 遍历(根集, False):
        try:
            with open(p, encoding="utf-8") as f:
                文 = f.read()
        except Exception:
            continue
        for t in 词模式.findall(文):
            计数[t] += 1
    return 计数


def 取定义(根集, 含类内):
    表 = []
    for p in 遍历(根集, True):
        try:
            with open(p, encoding="utf-8") as f:
                树 = ast.parse(f.read())
        except Exception:
            continue
        for n in 树.body:
            if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)):
                表.append((os.path.relpath(p, 根), n.lineno, n.name))
            elif isinstance(n, ast.ClassDef) and 含类内:
                for 子 in n.body:
                    if isinstance(子, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        表.append((os.path.relpath(p, 根), 子.lineno, 子.name))
    return 表


def 报(号, 定义根集, 引用根集, 含类内):
    计 = 文本计数(引用根集)
    定 = 取定义(定义根集, 含类内)
    零 = sorted(d for d in 定 if 计.get(d[2], 0) <= 1)
    print(f"【口径{号}】候选={len(定)} 零引用={len(零)}")
    with open(f"/tmp/清单_口径{号}.txt", "w", encoding="utf-8") as f:
        for 相对, 行, 名 in 零:
            f.write(f"{相对}:{行}\t{名}\n")
    return 零


报("①", 七大根, 七大根, False)
报("②", 六大根, 六大根, False)
报("③", 六大根, 六大根, True)
