### 2.3 「留痕」判定（三级，AST 级 + 源码级双判）

| 级别 | 判定 |
|---|---|
| `strong` 强留痕 | handler 体**内**存在调用，函数名命中日志/打印/写入/追加/上报/登记/审计/通知/快照类关键词；**或**存在 `raise`。AST 级，不看注释。 |
| `weak` 弱留痕 | handler 体**内**源码文本出现 `log/print/记录/追加/append/写入/落盘/上报/登记/审计/反馈/警告/失败/异常/error/warn/trace/快照` |

## 附录 A · 复算脚本源码（自包含，可直接粘贴运行）

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""宽泛异常捕获 AST 复算（只读）。单一 walk 入口 + finally/循环上下文 + 三级留痕。"""
import ast, json, os, io, re
from collections import Counter

ROOT = "/Users/hekunhua/Documents/Agent/PHP/系统工程平台"
EXCLUDE_DIRS = {"__pycache__", ".git", "工程缓存", "node_modules",
                ".venv", "venv", ".mypy_cache", "tmp"}

STRONG_FUNC_KW = ["log", "print", "warn", "error", "debug", "info", "critical",
                  "exception", "record", "report", "emit", "notify", "trace",
                  "审计", "记录", "写入", "追加", "上报", "登记", "落盘",
                  "推送", "通知", "反馈", "快照"]
STRONG_ATTR_KW = ["日志", "记录", "审计", "追踪", "失败", "异常"]
WEAK_KW = ["log", "print", "记录", "追加", "append", "写入", "落盘", "上报",
           "登记", "审计", "反馈", "警告", "失败", "异常",
           "error", "warn", "trace", "快照"]
CARRIES_RE = r'str\(错误\)|错误说明|错误码|失败原因|str\(异常\)|repr\(错误\)'


def collect(root):
    out = []
    for dp, dn, fn in os.walk(root):
        dn[:] = [d for d in dn if d not in EXCLUDE_DIRS]
        for f in fn:
            if f.endswith(".py"):
                out.append(os.path.relpath(os.path.join(dp, f), root))
    return sorted(out)


def kind_of(h):
    if h.type is None:
        return "裸except", []
    if isinstance(h.type, ast.Tuple):
        ns = [ast.unparse(e) for e in h.type.elts]
    else:
        ns = [ast.unparse(h.type)]
    if ns == ["Exception"]:
        return "Exception", ns
    if ns == ["BaseException"]:
        return "BaseException", ns
    if any(x in ("Exception", "BaseException") for x in ns):
        return "元组含宽泛", ns
    return "具体类型", ns


def strong(body):
    for st in body:
        for x in ast.walk(st):
            if isinstance(x, ast.Call):
                f = x.func
                nm = f.id if isinstance(f, ast.Name) else (
                    ast.unparse(f) if isinstance(f, ast.Attribute) else "")
                if any(k in nm.lower() for k in STRONG_FUNC_KW) or any(k in nm for k in STRONG_ATTR_KW):
                    return True
            if isinstance(x, ast.Raise):
                return True
    return False


def main():
    files = collect(ROOT)
    hits, errs = [], []
    for rel in files:
        p = os.path.join(ROOT, rel)
        try:
            src = io.open(p, encoding="utf-8", errors="replace").read()
            tree = ast.parse(src)
        except Exception as e:
            errs.append({"file": rel, "err": str(e)[:200]})
            continue

        st_fin, st_loop = [0], [0]

        def walk(n):
            if isinstance(n, ast.Try):
                for h in n.handlers:
                    k, ns = kind_of(h)
                    body = "\n".join(ast.get_source_segment(src, s) or "" for s in h.body)
                    s_tr = strong(h.body)
                    w_tr = (not s_tr) and any(kk in body for kk in WEAK_KW)
                    hits.append({
                        "file": rel, "line": h.lineno, "kind": k, "names": ns,
                        "level": "strong" if s_tr else ("weak" if w_tr else "none"),
                        "in_finally": st_fin[0] > 0, "in_loop": st_loop[0] > 0,
                        "silent": all(
                            isinstance(s, ast.Pass)
                            or (isinstance(s, ast.Expr) and isinstance(s.value, ast.Constant))
                            or isinstance(s, (ast.Continue, ast.Break))
                            for s in h.body),
                        "ret": bool(h.body) and all(isinstance(s, ast.Return) for s in h.body),
                        "body": body.strip()[:400],
                    })
                    for s in h.body:      # handler 体只在这里下探一次
                        walk(s)
                for s in n.body: walk(s)
                for s in n.orelse: walk(s)
                if n.finalbody:
                    st_fin[0] += 1
                    for s in n.finalbody: walk(s)
                    st_fin[0] -= 1
                return
            if isinstance(n, (ast.For, ast.While)):
                st_loop[0] += 1
                for c in ast.iter_child_nodes(n): walk(c)
                st_loop[0] -= 1
                return
            if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.Lambda, ast.ClassDef)):
                sv = st_loop[0]; st_loop[0] = 0
                for c in ast.iter_child_nodes(n): walk(c)
                st_loop[0] = sv
                return
            for c in ast.iter_child_nodes(n): walk(c)

        walk(tree)

    json.dump({"files": len(files), "errors": errs, "hits": hits},
              io.open("/tmp/宽泛异常_raw.json", "w", encoding="utf-8"),
              ensure_ascii=False, indent=1)

    print("扫描文件:", len(files), "解析失败:", len(errs))
    print("异常处理器总数:", len(hits))
    print("类型分布:", dict(Counter(h["kind"] for h in hits)))
    w = [h for h in hits if h["kind"] in ("Exception", "BaseException")]
    print("宽泛合计:", len(w),
          "Exception:", sum(1 for h in w if h["kind"] == "Exception"),
          "BaseException:", sum(1 for h in w if h["kind"] == "BaseException"))
    print("宽泛留痕级别:", dict(Counter(h["level"] for h in w)))
    none = [h for h in w if h["level"] == "none"]
    pure = [h for h in none if not re.search(CARRIES_RE, h["body"])]
    print("无留痕:", len(none), "准留痕:", len(none) - len(pure), "净静默:", len(pure))
    print("净静默形态: 静默", sum(1 for h in pure if h["silent"]),
          "仅return", sum(1 for h in pure if h["ret"]),
          "赋值默认", sum(1 for h in pure if not h["silent"] and not h["ret"]),
          "finally内", sum(1 for h in pure if h["in_finally"]))
    print("宽泛在finally上下文内:", sum(1 for h in w if h["in_finally"]),
          "其中有留痕:", sum(1 for h in w if h["in_finally"] and h["level"] != "none"))
    for k, v in Counter(h["file"].split("/")[0] for h in hits).most_common():
        print("   ", v, k)

    # 净静默里真静默的明细（P0 清单）
    print("\n--- 净静默中 pass/continue（P0）---")
    for h in sorted((x for x in pure if x["silent"]), key=lambda a: (a["file"], a["line"])):
        print(f"  {h['file']}:{h['line']}")


if __name__ == "__main__":
    main()
```
