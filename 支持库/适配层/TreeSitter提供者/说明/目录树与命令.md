# 命令

```bash
# 语法检查（单文件）
cd 项目根 && unset PYTHONPATH && /opt/homebrew/bin/python3.14 -m py_compile 支持库/适配层/TreeSitter提供者/实现/提供者.py

# 依赖直连审计（本提供者内 tree_sitter 导入合法；其它目录出现即违规）
/opt/homebrew/bin/python3.14 -c "from 运行核心.依赖防火墙 import 审计依赖; r=审计依赖(); print([f'{v.文件}:{v.行号}' for v in r.违规列表 if 'tree_sitter' in (v.目标 or '')])"

# 消费方行为回归（语法索引）
/opt/homebrew/bin/python3.14 -m unittest 测试中心.支持库.核心.测试_语法索引
```
